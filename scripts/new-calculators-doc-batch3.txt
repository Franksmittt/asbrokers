function calculateEverestStrategyComparison() { 
const investmentAmount = evGetNumber("evInvestmentAmount"); 
const dividendTaxRate = evGetNumber("evDividendTax") / 100; 
const years = evGetNumber("evYears"); 
 
 
const rate128 = evGetNumber("evRate128") / 100; 
const bonus128 = evGetNumber("evBonus128") / 100; 
const rate142 = evGetNumber("evRate142") / 100; 
const rate145 = evGetNumber("evRate145") / 100; 
 
 
const errorBox = document.getElementById("evError"); 
const resultsBox = document.getElementById("evResults"); 
 
 
errorBox.style.display = "none" 
resultsBox.style.display = "none" 
 
 
if ( 
isNaN(investmentAmount) || 
isNaN(dividendTaxRate) || 
isNaN(years) || 
isNaN(rate128) || 
isNaN(bonus128) || 
isNaN(rate142) || 
isNaN(rate145) 
) { 
errorBox.innerText = "Please complete all fields." 
errorBox.style.display = "block" 
return; 
} 
 
 
if (investmentAmount < 100000) { 
errorBox.innerText = "Minimum investment amount is R100,000." 
errorBox.style.display = "block" 
return; 
} 
 
 
if ( 
dividendTaxRate < 0 || 
dividendTaxRate >= 1 || 
years <= 0 || 
rate128 < 0 || 
bonus128 < 0 || 
rate142 < 0 || 
rate145 < 0 
) { 
errorBox.innerText = "Please check the tax, return and term assumptions." 
errorBox.style.display = "block" 
return; 
} 
 
 
/* 12.8% income option */ 
const grossAnnual128 = investmentAmount * rate128; 
const taxAnnual128 = grossAnnual128 * dividendTaxRate; 
const netAnnual128 = grossAnnual128 - taxAnnual128; 
const netMonthly128 = netAnnual128 / 12; 
const totalIncome128 = netAnnual128 * years; 
 
 
const grossBonus128 = years >= 5 ? investmentAmount * bonus128 : 0; 
const bonusTax128 = grossBonus128 * dividendTaxRate; 
const netBonus128 = grossBonus128 - bonusTax128; 
 
 
const capital128 = investmentAmount; 
const totalOutcome128 = capital128 + totalIncome128 + netBonus128; 
 
 
/* 14.2% income option */ 
const grossAnnual142 = investmentAmount * rate142; 
const taxAnnual142 = grossAnnual142 * dividendTaxRate; 
const netAnnual142 = grossAnnual142 - taxAnnual142; 
const netMonthly142 = netAnnual142 / 12; 
const totalIncome142 = netAnnual142 * years; 
 
 
const capital142 = investmentAmount; 
const totalOutcome142 = capital142 + totalIncome142; 
 
 
/* 14.5% growth option */ 
const grossValue145 = investmentAmount * Math.pow(1 + rate145, years); 
const grossGrowth145 = grossValue145 - investmentAmount; 
const taxGrowth145 = grossGrowth145 * dividendTaxRate; 
const netGrowth145 = grossGrowth145 - taxGrowth145; 
const maturityValue145 = investmentAmount + netGrowth145; 
 
 
const netMonthly145 = 0; 
const netAnnual145 = 0; 
const totalIncome145 = 0; 
const totalOutcome145 = maturityValue145; 
 
 
/* Winners */ 
const monthlyOptions = [ 
{ name: "12.8% Income + Bonus", value: netMonthly128 }, 
{ name: "14.2% Income", value: netMonthly142 }, 
{ name: "14.5% Growth", value: netMonthly145 } 
]; 
 
 
const totalOptions = [ 
{ name: "12.8% Income + Bonus", value: totalOutcome128 }, 
{ name: "14.2% Income", value: totalOutcome142 }, 
{ name: "14.5% Growth", value: totalOutcome145 } 
]; 
 
 
monthlyOptions.sort((a, b) => b.value - a.value); 
totalOptions.sort((a, b) => b.value - a.value); 
 
 
const monthlyWinner = monthlyOptions[0]; 
const totalWinner = totalOptions[0]; 
 
 
document.getElementById("evMonthlyWinner").innerText = monthlyWinner.name; 
document.getElementById("evMonthlyWinnerText").innerText = 
monthlyWinner.name + " provides the highest monthly income at approximately " + 
evFormatRand(monthlyWinner.value) + " per month." 
 
 
document.getElementById("evTotalWinner").innerText = totalWinner.name; 
document.getElementById("evTotalWinnerText").innerText = 
totalWinner.name + " produces the highest projected total outcome over " + 
years + " years at approximately " + evFormatRand(totalWinner.value) + 
", including capital where applicable." 
 
 
/* Fill table */ 
document.getElementById("evGrossRate128").innerText = (rate128 * 100).toFixed(1) + "%" 
document.getElementById("evGrossRate142").innerText = (rate142 * 100).toFixed(1) + "%" 
document.getElementById("evGrossRate145").innerText = (rate145 * 100).toFixed(1) + "% accumulated" 
 
 
document.getElementById("evMonthly128").innerText = evFormatRand(netMonthly128); 
document.getElementById("evMonthly142").innerText = evFormatRand(netMonthly142); 
document.getElementById("evMonthly145").innerText = "R 0" 
 
 
document.getElementById("evAnnual128").innerText = evFormatRand(netAnnual128); 
document.getElementById("evAnnual142").innerText = evFormatRand(netAnnual142); 
document.getElementById("evAnnual145").innerText = "R 0" 
 
 
document.getElementById("evIncomePaid128").innerText = evFormatRand(totalIncome128); 
document.getElementById("evIncomePaid142").innerText = evFormatRand(totalIncome142); 
document.getElementById("evIncomePaid145").innerText = "R 0" 
 
 
document.getElementById("evBonusNet128").innerText = evFormatRand(netBonus128); 
document.getElementById("evGrowthNet145").innerText = evFormatRand(netGrowth145); 
 
 
document.getElementById("evCapital128").innerText = evFormatRand(capital128); 
document.getElementById("evCapital142").innerText = evFormatRand(capital142); 
document.getElementById("evCapital145").innerText = evFormatRand(maturityValue145); 
 
 
document.getElementById("evTotal128").innerText = evFormatRand(totalOutcome128); 
document.getElementById("evTotal142").innerText = evFormatRand(totalOutcome142); 
document.getElementById("evTotal145").innerText = evFormatRand(totalOutcome145); 
 
 
document.getElementById("evSummaryText").innerText = 
"If the investor needs monthly income, the 14.2% income option produces the highest income from day one. " + 
"If the investor wants income but is willing to accept lower monthly cash flow in exchange for a five-year bonus, " + 
"the 12.8% option may produce a stronger total income result over five years. " + 
"If the investor does not need income and wants to maximise the projected maturity value, the 14.5% growth option is designed for capital accumulation. " + 
"The correct strategy depends on income need, time horizon, liquidity requirement, tax position, product risk and suitability." 
 
 
resultsBox.style.display = "block" 
}

function calculateEverestLivingAnnuity() { 
const capital = everestLaGetNumber("everestLaCapital"); 
const drawdownRate = everestLaGetNumber("everestLaDrawdown") / 100; 
const returnRate = everestLaGetNumber("everestLaReturn") / 100; 
const inflationDividendRate = everestLaGetNumber("everestLaInflationDividend") / 100; 
 
 
const errorBox = document.getElementById("everestLaError"); 
const resultsBox = document.getElementById("everestLaResults"); 
 
 
errorBox.style.display = "none" 
resultsBox.style.display = "none" 
 
 
if ( 
isNaN(capital) || 
isNaN(drawdownRate) || 
isNaN(returnRate) || 
isNaN(inflationDividendRate) 
) { 
errorBox.innerText = "Please complete all fields." 
errorBox.style.display = "block" 
return; 
} 
 
 
if (capital <= 0) { 
errorBox.innerText = "Living annuity capital must be greater than zero." 
errorBox.style.display = "block" 
return; 
} 
 
 
if (drawdownRate < 0.025 || drawdownRate > 0.175) { 
errorBox.innerText = "Living annuity drawdown must be between 2.5% and 17.5%." 
errorBox.style.display = "block" 
return; 
} 
 
 
if (returnRate < 0) { 
errorBox.innerText = "Targeted annual return cannot be negative." 
errorBox.style.display = "block" 
return; 
} 
 
 
if (inflationDividendRate < 0) { 
errorBox.innerText = "Inflation protection dividend cannot be negative." 
errorBox.style.display = "block" 
return; 
} 
 
 
const annualIncome = capital * drawdownRate; 
const monthlyIncome = annualIncome / 12; 
 
 
const annualReturnAmount = capital * returnRate; 
const monthlyReturnAmount = annualReturnAmount / 12; 
 
 
const annualSurplus = annualReturnAmount - annualIncome; 
const surplusPercent = returnRate - drawdownRate; 
 
 
const inflationDividendAmount = capital * inflationDividendRate; 
 
 
const fiveYearIncome = annualIncome * 5; 
const fiveYearTotal = fiveYearIncome + inflationDividendAmount; 
 
 
document.getElementById("everestLaMonthlyIncome").innerText = 
everestLaFormatRand(monthlyIncome); 
 
 
document.getElementById("everestLaAnnualIncome").innerText = 
everestLaFormatRand(annualIncome); 
 
 
document.getElementById("everestLaSurplus").innerText = 
everestLaFormatRand(annualSurplus); 
 
 
document.getElementById("everestLaAnnualReturnAmount").innerText = 
everestLaFormatRand(annualReturnAmount); 
 
 
document.getElementById("everestLaMonthlyReturnAmount").innerText = 
everestLaFormatRand(monthlyReturnAmount); 
 
 
document.getElementById("everestLaSurplusPercent").innerText = 
(surplusPercent * 100).toFixed(2) + "%" 
 
 
document.getElementById("everestLaInflationDividendAmount").innerText = 
everestLaFormatRand(inflationDividendAmount); 
 
 
document.getElementById("everestLaFiveYearIncome").innerText = 
everestLaFormatRand(fiveYearIncome); 
 
 
document.getElementById("everestLaFiveYearTotal").innerText = 
everestLaFormatRand(fiveYearTotal); 
 
 
let surplusMessage = "" 
 
 
if (annualSurplus > 0) { 
surplusMessage = 
"The targeted annual return is higher than the selected drawdown, leaving an illustrative surplus of " + 
everestLaFormatRand(annualSurplus) + 
" per year before fees, tax, product rules and annual review conditions." 
} else if (annualSurplus === 0) { 
surplusMessage = 
"The selected drawdown is equal to the targeted annual return. This leaves no illustrative surplus before fees, tax, product rules and annual review conditions." 
} else { 
surplusMessage = 
"The selected drawdown is higher than the targeted annual return. This creates an illustrative shortfall of " + 
everestLaFormatRand(Math.abs(annualSurplus)) + 
" per year before fees, tax, product rules and annual review conditions." 
} 
 
 
document.getElementById("everestLaSummaryText").innerText = 
"Based on living annuity capital of " + everestLaFormatRand(capital) + 
", a selected drawdown rate of " + (drawdownRate * 100).toFixed(1) + 
"% produces estimated gross income of " + everestLaFormatRand(monthlyIncome) + 
" per month, or " + everestLaFormatRand(annualIncome) + 
" per year. At a targeted annual return of " + (returnRate * 100).toFixed(2) + 
"%, the illustrative annual return amount is " + everestLaFormatRand(annualReturnAmount) + 
". " + surplusMessage + 
" The illustrative five-year inflation protection dividend is " + 
everestLaFormatRand(inflationDividendAmount) + 
", based on " + (inflationDividendRate * 100).toFixed(1) + 
"% of original capital. This calculator does not calculate personal income tax and does not confirm suitability." 
 
 
resultsBox.style.display = "block" 
}

function calculateAverageClause() { 
const sumInsured = Number(document.getElementById("sumInsured").value) || 0; 
const actualValue = Number(document.getElementById("actualValue").value) || 0;

function calculateGrowth() { 
const symbol = document.getElementById("currency").value; 
const initial = Number(document.getElementById("initialInvestment").value) || 0; 
const years = Math.max(Number(document.getElementById("years").value) || 0, 0); 
const monthlyContributionStart = Number(document.getElementById("monthlyContribution").value) || 0; 
const annualIncrease = (Number(document.getElementById("annualIncrease").value) || 0) / 100; 
const annualGrowth = (Number(document.getElementById("growthRate").value) || 0) / 100; 
 
 
let balance = initial; 
let totalInvested = initial; 
let monthlyContribution = monthlyContributionStart; 
const monthlyGrowthRate = Math.pow(1 + annualGrowth, 1 / 12) - 1; 
 
 
for (let year = 1; year <= years; year++) { 
for (let month = 1; month <= 12; month++) { 
balance = balance * (1 + monthlyGrowthRate); 
balance += monthlyContribution; 
totalInvested += monthlyContribution; 
} 
 
 
monthlyContribution = monthlyContribution * (1 + annualIncrease); 
} 
 
 
const growthEarned = balance - totalInvested; 
 
 
document.getElementById("futureValue").textContent = formatMoney(balance, symbol); 
document.getElementById("totalInvested").textContent = formatMoney(totalInvested, symbol); 
document.getElementById("growthEarned").textContent = formatMoney(growthEarned, symbol); 
}

function money(value) { 
return "R " + Number(value).toLocaleString("en-ZA", { 
minimumFractionDigits: 2, 
maximumFractionDigits: 2 
}); 
} 
 
 
function percent(value) { 
return (value * 100).toFixed(4) + "%" 
} 
 
 
function projectValue(start, months, contribution, annualIncrease, monthlyRate) { 
let value = start; 
let totalContributions = 0; 
let rows = []; 
 
 
for (let month = 1; month <= months; month++) { 
let opening = value; 
 
 
let contributionFactor = Math.pow( 
1 + annualIncrease, 
Math.floor((month - 1) / 12) 
); 
 
 
let monthContribution = contribution * contributionFactor; 
 
 
value += monthContribution; 
totalContributions += monthContribution; 
 
 
let growthEarned = value * monthlyRate; 
value += growthEarned; 
 
 
rows.push({ 
month, 
opening, 
contribution: monthContribution, 
growth: growthEarned, 
closing: value 
}); 
} 
 
 
return { 
finalValue: value, 
totalContributions, 
rows 
}; 
} 
 
 
function solveRequiredMonthlyRate(start, target, months, contribution, annualIncrease) { 
let low = -0.99; 
let high = 10; 
 
 
for (let i = 0; i < 200; i++) { 
let mid = (low + high) / 2; 
let result = projectValue(start, months, contribution, annualIncrease, mid); 
 
 
if (result.finalValue < target) { 
low = mid; 
} else { 
high = mid; 
} 
} 
 
 
return (low + high) / 2; 
} 
 
 
function calculateGoal() { 
const start = Number(document.getElementById("startingValue").value); 
const target = Number(document.getElementById("targetValue").value); 
const months = Number(document.getElementById("months").value); 
const contribution = Number(document.getElementById("monthlyContribution").value); 
const annualIncrease = Number(document.getElementById("annualIncrease").value) / 100; 
const mode = document.getElementById("growthMode").value; 
 
 
if (start <= 0 || months <= 0) { 
alert("Starting value and number of months must be greater than zero."); 
return; 
} 
 
 
let monthlyRate; 
 
 
if (mode === "required") { 
monthlyRate = solveRequiredMonthlyRate( 
start, 
target, 
months, 
contribution, 
annualIncrease 
); 
} else { 
monthlyRate = Number(document.getElementById("manualMonthlyRate").value) / 100; 
} 
 
 
const projection = projectValue( 
start, 
months, 
contribution, 
annualIncrease, 
monthlyRate 
); 
 
 
const annualRate = Math.pow(1 + monthlyRate, 12) - 1; 
const growthCreated = projection.finalValue - start - projection.totalContributions; 
 
 
document.getElementById("monthlyRateResult").innerText = percent(monthlyRate); 
document.getElementById("annualRateResult").innerText = percent(annualRate); 
document.getElementById("finalValueResult").innerText = money(projection.finalValue); 
document.getElementById("totalContributionsResult").innerText = money(projection.totalContributions); 
document.getElementById("growthCreatedResult").innerText = money(growthCreated); 
document.getElementById("monthsResult").innerText = months; 
 
 
currentRows = projection.rows; 
renderTable(currentRows); 
} 
 
 
function renderTable(rows) { 
const tbody = document.getElementById("resultTable"); 
tbody.innerHTML = "" 
 
 
rows.forEach(row => { 
const tr = document.createElement("tr"); 
 
 
tr.innerHTML = ` 
 ${row.month} 
 ${money(row.opening)} 
 ${money(row.contribution)} 
 ${money(row.growth)} 
 ${money(row.closing)} 
`; 
 
 
tbody.appendChild(tr); 
}); 
} 
 
 
function downloadCSV() { 
if (!currentRows.length) { 
alert("Please calculate the goal first."); 
return; 
} 
 
 
let csv = "Month,Opening Value,Monthly Contribution,Growth Earned,Closing Value\n" 
 
 
currentRows.forEach(row => {

].join(",") + "\n" 
});

a.href = url; 
a.download = "personal-goal-growth-calculator.csv" 
a.click(); 
 
 
window.URL.revokeObjectURL(url); 
}

- id uuid PK, content text, author_id → auth.users, created_at timestamptz

- id uuid PK, lead_id → crm_leads ON DELETE CASCADE
- title varchar(255), due_date timestamptz, is_completed boolean default false

- id uuid PK, lead_id → crm_leads ON DELETE CASCADE (nullable)

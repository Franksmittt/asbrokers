const investmentAmount = evGetNumber("evInvestmentAmount");

const dividendTaxRate = evGetNumber("evDividendTax") / 100;

const years = evGetNumber("evYears");

const rate128 = evGetNumber("evRate128") / 100;

const bonus128 = evGetNumber("evBonus128") / 100;

const rate142 = evGetNumber("evRate142") / 100;

const rate145 = evGetNumber("evRate145") / 100;

const errorBox = document.getElementById("evError");

const resultsBox = document.getElementById("evResults");

errorBox.style.display = "none"

resultsBox.style.display = "none"

if (

isNaN(investmentAmount) ||

isNaN(dividendTaxRate) ||

isNaN(years) ||

isNaN(rate128) ||

isNaN(bonus128) ||

isNaN(rate142) ||

isNaN(rate145)

) {

errorBox.innerText = "Please complete all fields."

errorBox.style.display = "block"

return;

}

if (investmentAmount < 100000) {

errorBox.innerText = "Minimum investment amount is R100,000."

errorBox.style.display = "block"

return;

}

if (

dividendTaxRate < 0 ||

dividendTaxRate >= 1 ||

years <= 0 ||

rate128 < 0 ||

bonus128 < 0 ||

rate142 < 0 ||

rate145 < 0

) {

errorBox.innerText = "Please check the tax, return and term assumptions."

errorBox.style.display = "block"

return;

}

/* 12.8% income option */

const grossAnnual128 = investmentAmount * rate128;

const taxAnnual128 = grossAnnual128 * dividendTaxRate;

const netAnnual128 = grossAnnual128 - taxAnnual128;

const netMonthly128 = netAnnual128 / 12;

const totalIncome128 = netAnnual128 * years;

const grossBonus128 = years >= 5 ? investmentAmount * bonus128 : 0;

const bonusTax128 = grossBonus128 * dividendTaxRate;

const netBonus128 = grossBonus128 - bonusTax128;

const capital128 = investmentAmount;

const totalOutcome128 = capital128 + totalIncome128 + netBonus128;

/* 14.2% income option */

const grossAnnual142 = investmentAmount * rate142;

const taxAnnual142 = grossAnnual142 * dividendTaxRate;

const netAnnual142 = grossAnnual142 - taxAnnual142;

const netMonthly142 = netAnnual142 / 12;

const totalIncome142 = netAnnual142 * years;

const capital142 = investmentAmount;

const totalOutcome142 = capital142 + totalIncome142;

/* 14.5% growth option */

const grossValue145 = investmentAmount * Math.pow(1 + rate145, years);

const grossGrowth145 = grossValue145 - investmentAmount;

const taxGrowth145 = grossGrowth145 * dividendTaxRate;

const netGrowth145 = grossGrowth145 - taxGrowth145;

const maturityValue145 = investmentAmount + netGrowth145;

const netMonthly145 = 0;

const netAnnual145 = 0;

const totalIncome145 = 0;

const totalOutcome145 = maturityValue145;

/* Winners */

const monthlyOptions = [

{ name: "12.8% Income + Bonus", value: netMonthly128 },

{ name: "14.2% Income", value: netMonthly142 },

{ name: "14.5% Growth", value: netMonthly145 }

];

const totalOptions = [

{ name: "12.8% Income + Bonus", value: totalOutcome128 },

{ name: "14.2% Income", value: totalOutcome142 },

{ name: "14.5% Growth", value: totalOutcome145 }

];

monthlyOptions.sort((a, b) => b.value - a.value);

totalOptions.sort((a, b) => b.value - a.value);

const monthlyWinner = monthlyOptions[0];

const totalWinner = totalOptions[0];

document.getElementById("evMonthlyWinner").innerText = monthlyWinner.name;

document.getElementById("evMonthlyWinnerText").innerText =

monthlyWinner.name + " provides the highest monthly income at approximately " +

evFormatRand(monthlyWinner.value) + " per month."

document.getElementById("evTotalWinner").innerText = totalWinner.name;

document.getElementById("evTotalWinnerText").innerText =

totalWinner.name + " produces the highest projected total outcome over " +

years + " years at approximately " + evFormatRand(totalWinner.value) +

", including capital where applicable."

/* Fill table */

document.getElementById("evGrossRate128").innerText = (rate128 * 100).toFixed(1) + "%"

document.getElementById("evGrossRate142").innerText = (rate142 * 100).toFixed(1) + "%"

document.getElementById("evGrossRate145").innerText = (rate145 * 100).toFixed(1) + "% accumulated"

document.getElementById("evMonthly128").innerText = evFormatRand(netMonthly128);

document.getElementById("evMonthly142").innerText = evFormatRand(netMonthly142);

document.getElementById("evMonthly145").innerText = "R 0"

document.getElementById("evAnnual128").innerText = evFormatRand(netAnnual128);

document.getElementById("evAnnual142").innerText = evFormatRand(netAnnual142);

document.getElementById("evAnnual145").innerText = "R 0"

document.getElementById("evIncomePaid128").innerText = evFormatRand(totalIncome128);

document.getElementById("evIncomePaid142").innerText = evFormatRand(totalIncome142);

document.getElementById("evIncomePaid145").innerText = "R 0"

document.getElementById("evBonusNet128").innerText = evFormatRand(netBonus128);

document.getElementById("evGrowthNet145").innerText = evFormatRand(netGrowth145);

document.getElementById("evCapital128").innerText = evFormatRand(capital128);

document.getElementById("evCapital142").innerText = evFormatRand(capital142);

document.getElementById("evCapital145").innerText = evFormatRand(maturityValue145);

document.getElementById("evTotal128").innerText = evFormatRand(totalOutcome128);

document.getElementById("evTotal142").innerText = evFormatRand(totalOutcome142);

document.getElementById("evTotal145").innerText = evFormatRand(totalOutcome145);

document.getElementById("evSummaryText").innerText =

"If the investor needs monthly income, the 14.2% income option produces the highest income from day one. " +

"If the investor wants income but is willing to accept lower monthly cash flow in exchange for a five-year bonus, " +

"the 12.8% option may produce a stronger total income result over five years. " +

"The correct strategy depends on income need, time horizon, liquidity requirement, tax position, product risk and suitability."

resultsBox.style.display = "block"

}

function calculateEverestLivingAnnuity() {

const capital = everestLaGetNumber("everestLaCapital");

const drawdownRate = everestLaGetNumber("everestLaDrawdown") / 100;

const returnRate = everestLaGetNumber("everestLaReturn") / 100;

const inflationDividendRate = everestLaGetNumber("everestLaInflationDividend") / 100;

const errorBox = document.getElementById("everestLaError");

const resultsBox = document.getElementById("everestLaResults");

errorBox.style.display = "none"

resultsBox.style.display = "none"

if (

isNaN(capital) ||

isNaN(drawdownRate) ||

isNaN(returnRate) ||

isNaN(inflationDividendRate)

) {

errorBox.innerText = "Please complete all fields."

errorBox.style.display = "block"

return;

}

if (capital <= 0) {

errorBox.innerText = "Living annuity capital must be greater than zero."

errorBox.style.display = "block"

return;

}

if (drawdownRate < 0.025 || drawdownRate > 0.175) {

errorBox.innerText = "Living annuity drawdown must be between 2.5% and 17.5%."

errorBox.style.display = "block"

return;

}

if (returnRate < 0) {

errorBox.innerText = "Targeted annual return cannot be negative."

errorBox.style.display = "block"

return;

}

if (inflationDividendRate < 0) {

errorBox.innerText = "Inflation protection dividend cannot be negative."

errorBox.style.display = "block"

return;

}

const annualIncome = capital * drawdownRate;

const monthlyIncome = annualIncome / 12;

function money(value) {

function percent(value) {

return (value * 100).toFixed(4) + "%"

function projectValue(start, months, contribution, annualIncrease, monthlyRate) {

let value = start;

let totalContributions = 0;

let rows = [];

for (let month = 1; month <= months; month++) {

let opening = value;

let contributionFactor = Math.pow(

1 + annualIncrease,

Math.floor((month - 1) / 12)

);

let monthContribution = contribution * contributionFactor;

value += monthContribution;

totalContributions += monthContribution;

let growthEarned = value * monthlyRate;

value += growthEarned;

rows.push({

month,

opening,

contribution: monthContribution,

growth: growthEarned,

closing: value

});

}

return {

finalValue: value,

totalContributions,

rows

};

}

function solveRequiredMonthlyRate(start, target, months, contribution, annualIncrease) {

let low = -0.99;

let high = 10;

for (let i = 0; i < 200; i++) {

let mid = (low + high) / 2;

let result = projectValue(start, months, contribution, annualIncrease, mid);

if (result.finalValue < target) {

low = mid;

} else {

high = mid;

}

}

return (low + high) / 2;

}

function estateFormatRand(value) {
return "R " + Math.round(value).toLocaleString("en-ZA");
}


function estateGetNumber(id) {
return parseFloat(document.getElementById(id).value);

function calculateEstateDutyAmount(dutiableValue) {
if (dutiableValue <= 0) {
return 0;
}


if (dutiableValue <= 30000000) {
return dutiableValue * 0.20;
}


return (30000000 * 0.20) + ((dutiableValue - 30000000) * 0.25);
}


function calculateEstateDuty() {
const grossEstateValue = estateGetNumber("estateGrossValue");
const liabilities = estateGetNumber("estateLiabilities");
const spouseBequest = estateGetNumber("estateSpouseBequest");


const errorBox = document.getElementById("estateError");
const resultsBox = document.getElementById("estateResults");


errorBox.style.display = "none"
resultsBox.style.display = "none"


if (
isNaN(grossEstateValue) ||
isNaN(liabilities) ||
isNaN(spouseBequest)
) {
errorBox.innerText = "Please complete all fields."
errorBox.style.display = "block"
return;
}


if (grossEstateValue < 0 || liabilities < 0 || spouseBequest < 0) {
errorBox.innerText = "Values cannot be negative."
errorBox.style.display = "block"
return;
}


const section4AAbatement = 3500000;


const netEstateBeforeAbatement = Math.max(
0,
grossEstateValue - liabilities - spouseBequest
);


const abatementUsed = Math.min(section4AAbatement, netEstateBeforeAbatement);
const dutiableEstateValue = Math.max(0, netEstateBeforeAbatement - section4AAbatement);


const estateDutyPayable = calculateEstateDutyAmount(dutiableEstateValue);


const executorFeeBeforeVat = grossEstateValue * 0.035;
const executorVat = executorFeeBeforeVat * 0.15;
const executorFees = executorFeeBeforeVat + executorVat;


const totalEstateCosts = estateDutyPayable + executorFees;


document.getElementById("estateTotalCosts").innerText = estateFormatRand(totalEstateCosts);
document.getElementById("estateDutyPayable").innerText = estateFormatRand(estateDutyPayable);
document.getElementById("estateExecutorFees").innerText = estateFormatRand(executorFees);
document.getElementById("estateNetBeforeAbatement").innerText = estateFormatRand(netEstateBeforeAbatement);
document.getElementById("estateAbatementUsed").innerText = estateFormatRand(abatementUsed);
document.getElementById("estateDutiableValue").innerText = estateFormatRand(dutiableEstateValue);


document.getElementById("estateSummaryText").innerText =
"Based on a gross estate value of " + estateFormatRand(grossEstateValue) +
", liabilities of " + estateFormatRand(liabilities) +
" and spouse bequests of " + estateFormatRand(spouseBequest) +
", the estimated net estate before abatement is " +
estateFormatRand(netEstateBeforeAbatement) +
". After applying the Section 4A abatement, the estimated dutiable estate value is " +
estateFormatRand(dutiableEstateValue) +
". Estimated estate duty is " + estateFormatRand(estateDutyPayable) +
". Estimated executor fees are " + estateFormatRand(executorFees) +
". This means the estate may need approximately " +
estateFormatRand(totalEstateCosts) +
" in cash for these two estate costs alone. Other estate costs may still apply."


resultsBox.style.display = "block"
}

function reductionFormatRand(value) {
return "R " + Math.round(value).toLocaleString("en-ZA");
}


function reductionGetNumber(id) {
return parseFloat(document.getElementById(id).value);
}


function calculateEstateReduction() {
const annualDonation = reductionGetNumber("reductionDonationType");
const currentAge = reductionGetNumber("reductionCurrentAge");
const plannedAge = reductionGetNumber("reductionPlannedAge");
const growthRate = reductionGetNumber("reductionGrowthRate") / 100;


const errorBox = document.getElementById("reductionError");
const resultsBox = document.getElementById("reductionResults");


errorBox.style.display = "none"
resultsBox.style.display = "none"


if (
isNaN(annualDonation) ||
isNaN(currentAge) ||
isNaN(plannedAge) ||
isNaN(growthRate)
) {
errorBox.innerText = "Please complete all fields."
errorBox.style.display = "block"
return;
}


if (plannedAge <= currentAge) {
errorBox.innerText = "Planned age must be higher than current age."
errorBox.style.display = "block"
return;
}


if (growthRate < -0.99) {
errorBox.innerText = "Growth rate is too low."
errorBox.style.display = "block"
return;
}


const donationPeriod = plannedAge - currentAge;
const totalDonations = annualDonation * donationPeriod;
const estimatedEstateDutySaving = totalDonations * 0.20;


let estimatedTrustValue = 0;


for (let year = 1; year <= donationPeriod; year++) {
estimatedTrustValue = estimatedTrustValue * (1 + growthRate);
estimatedTrustValue += annualDonation;
}


const donorTypeText = annualDonation === 300000 ? "a couple" : "a single person"


document.getElementById("reductionEstateDutySaving").innerText =
reductionFormatRand(estimatedEstateDutySaving);


document.getElementById("reductionTotalDonations").innerText =
reductionFormatRand(totalDonations);


document.getElementById("reductionTrustValue").innerText =
reductionFormatRand(estimatedTrustValue);


document.getElementById("reductionAnnualDonation").innerText =
reductionFormatRand(annualDonation);


document.getElementById("reductionDonationPeriod").innerText =
donationPeriod + " years"


document.getElementById("reductionSummaryText").innerText =
"Based on " + donorTypeText +
" donating " + reductionFormatRand(annualDonation) +
" per year from age " + currentAge +
" to age " + plannedAge +
", the total donations made would be approximately " +
reductionFormatRand(totalDonations) +
". If those donations reduce the taxable estate, the illustrative estate duty saving at 20% could be about " +
reductionFormatRand(estimatedEstateDutySaving) +
". If the donated money is invested and grows at " +
(growthRate * 100).toFixed(1) +
"% per year, the illustrative investment value at age " +
plannedAge +
" could be approximately " +
reductionFormatRand(estimatedTrustValue) +
"."


resultsBox.style.display = "block"
}
</script>

function growthFormatRand(value) {
return "R " + Math.round(value).toLocaleString("en-ZA");

function growthGetNumber(id) {
return parseFloat(document.getElementById(id).value);

function calculateStrategicGrowth() {
const investmentAmount = growthGetNumber("growthInvestmentAmount");
const annualReturn = growthGetNumber("growthAnnualReturn") / 100;
const years = growthGetNumber("growthYears");
const dividendTaxRate = growthGetNumber("growthDividendTax") / 100;


const errorBox = document.getElementById("growthError");
const resultsBox = document.getElementById("growthResults");
const yearTable = document.getElementById("growthYearTable");


errorBox.style.display = "none"
resultsBox.style.display = "none"
yearTable.innerHTML = ""


if (
isNaN(investmentAmount) ||
isNaN(annualReturn) ||
isNaN(years) ||
isNaN(dividendTaxRate)
) {
errorBox.innerText = "Please complete all fields."
errorBox.style.display = "block"
return;
}


if (investmentAmount < 100000) {
errorBox.innerText = "Minimum investment amount is R100,000."
errorBox.style.display = "block"
return;
}


if (annualReturn < 0 || years <= 0 || dividendTaxRate < 0 || dividendTaxRate >= 1) {
errorBox.innerText = "Please check your return, term and tax assumptions."
errorBox.style.display = "block"
return;
}


const grossValue = investmentAmount * Math.pow(1 + annualReturn, years);
const grossGrowth = grossValue - investmentAmount;
const taxOnGrowth = grossGrowth * dividendTaxRate;
const netGrowth = grossGrowth - taxOnGrowth;
const projectedValue = investmentAmount + netGrowth;
const netReturnPercent = (netGrowth / investmentAmount) * 100;


document.getElementById("growthProjectedValue").innerText = growthFormatRand(projectedValue);
document.getElementById("growthCapitalInvested").innerText = growthFormatRand(investmentAmount);
document.getElementById("growthNetGrowth").innerText = growthFormatRand(netGrowth);
document.getElementById("growthGrossValue").innerText = growthFormatRand(grossValue);
document.getElementById("growthNetReturnPercent").innerText = netReturnPercent.toFixed(1) + "%"


document.getElementById("growthTaxSummary").innerHTML =
"Gross accumulated growth: <strong>" + growthFormatRand(grossGrowth) + "</strong><br>" +
"Less dividends tax on growth: <strong>− " + growthFormatRand(taxOnGrowth) + "</strong><br>" +
"Net growth kept by investor: <strong>" + growthFormatRand(netGrowth) + "</strong>"


for (let year = 1; year <= years; year++) {
const yearValue = investmentAmount * Math.pow(1 + annualReturn, year);
const yearGrowth = yearValue - investmentAmount;


const row = document.createElement("tr");
row.innerHTML =
"<td>Year " + year + "</td>" +
"<td>" + growthFormatRand(yearValue) + "</td>" +
"<td>" + growthFormatRand(yearGrowth) + "</td>"


yearTable.appendChild(row);
}


document.getElementById("growthSummaryText").innerText =
"Based on an investment of " + growthFormatRand(investmentAmount) +
", an annual accumulated return assumption of " + (annualReturn * 100).toFixed(1) +
"% over " + years + " years would produce a gross maturity value of " +
growthFormatRand(grossValue) + ". After applying dividends tax of " +
(dividendTaxRate * 100).toFixed(1) + "% to the accumulated growth, the projected value at maturity is " +
growthFormatRand(projectedValue) + ". This strategy is designed for investors who want capital growth rather than monthly income, subject to product risk, liquidity, tax and suitability."


resultsBox.style.display = "block"
}

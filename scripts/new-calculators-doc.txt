function growthFormatRand(value) {
return "R " + Math.round(value).toLocaleString("en-ZA");

function growthGetNumber(id) {
return parseFloat(document.getElementById(id).value);

function calculateRequiredGrowthRate() {
const currentAge = growthGetNumber("growthCurrentAge");
const retirementAge = growthGetNumber("growthRetirementAge");
const planningAge = growthGetNumber("growthPlanningAge");
const monthlyIncomeToday = growthGetNumber("growthMonthlyIncome");


const inflationRate = growthGetNumber("growthInflationRate") / 100;
const currentCapital = growthGetNumber("growthCurrentCapital");
const monthlySavings = growthGetNumber("growthMonthlySavings");
const savingsIncrease = growthGetNumber("growthSavingsIncrease") / 100;
const growthDuringRetirement = growthGetNumber("growthDuringRetirement") / 100;
const incomeIncreaseRetirement = growthGetNumber("growthIncomeIncreaseRetirement") / 100;


const errorBox = document.getElementById("growthError");
const resultsBox = document.getElementById("growthResults");


errorBox.style.display = "none"
resultsBox.style.display = "none"


if (
isNaN(currentAge) ||
isNaN(retirementAge) ||
isNaN(planningAge) ||
isNaN(monthlyIncomeToday) ||
isNaN(inflationRate) ||
isNaN(currentCapital) ||
isNaN(monthlySavings) ||
isNaN(savingsIncrease) ||
isNaN(growthDuringRetirement) ||
isNaN(incomeIncreaseRetirement)
) {
errorBox.innerText = "Please complete all fields."
errorBox.style.display = "block"
return;
}


if (retirementAge <= currentAge) {
errorBox.innerText = "Retirement age must be higher than current age."
errorBox.style.display = "block"
return;
}


if (planningAge <= retirementAge) {
errorBox.innerText = "Planning age must be higher than retirement age."
errorBox.style.display = "block"
return;
}


if (monthlyIncomeToday <= 0) {
errorBox.innerText = "Monthly income must be greater than zero."
errorBox.style.display = "block"
return;
}


if (currentCapital < 0 || monthlySavings < 0) {
errorBox.innerText = "Current capital and monthly savings cannot be negative."
errorBox.style.display = "block"
return;
}


const yearsToRetirement = retirementAge - currentAge;
const yearsInRetirement = planningAge - retirementAge;
const monthsToRetirement = yearsToRetirement * 12;
const monthsInRetirement = yearsInRetirement * 12;


const futureMonthlyIncome = monthlyIncomeToday * Math.pow(1 + inflationRate, yearsToRetirement);
const retirementMonthlyGrowth = Math.pow(1 + growthDuringRetirement, 1 / 12) - 1;


let capitalRequired = 0;


for (let month = 1; month <= monthsInRetirement; month++) {
const yearIndex = Math.floor((month - 1) / 12);
const monthlyIncomeAtMonth = futureMonthlyIncome * Math.pow(1 + incomeIncreaseRetirement, yearIndex);
capitalRequired += monthlyIncomeAtMonth / Math.pow(1 + retirementMonthlyGrowth, month);
}


function projectedCapitalAtGrowth(annualGrowthRate) {
const monthlyGrowthRate = Math.pow(1 + annualGrowthRate, 1 / 12) - 1;
let balance = currentCapital;


for (let month = 1; month <= monthsToRetirement; month++) {
const yearIndex = Math.floor((month - 1) / 12);
const escalatedSaving = monthlySavings * Math.pow(1 + savingsIncrease, yearIndex);


balance = balance * (1 + monthlyGrowthRate);
balance += escalatedSaving;
}


return balance;
}


let low = -0.9999;
let high = 1;


while (projectedCapitalAtGrowth(high) < capitalRequired) {
high = high * 2;


if (high > 1000) {
errorBox.innerText =
"The required growth rate is extremely high. This usually means the retirement target is too large for the current time period, capital and monthly savings."
errorBox.style.display = "block"
return;
}
}


for (let i = 0; i < 140; i++) {
const mid = (low + high) / 2;


if (projectedCapitalAtGrowth(mid) < capitalRequired) {
low = mid;
} else {
high = mid;
}
}


const requiredGrowthRate = high * 100;


document.getElementById("requiredGrowthRate").innerText = requiredGrowthRate.toFixed(2) + "% p.a."
document.getElementById("growthCapitalRequired").innerText = growthFormatRand(capitalRequired);
document.getElementById("growthFutureMonthlyIncome").innerText = growthFormatRand(futureMonthlyIncome);
document.getElementById("growthYearsToRetirement").innerText = yearsToRetirement + " years"
document.getElementById("growthIncomePlanningPeriod").innerText =
retirementAge + " to " + planningAge + " (" + yearsInRetirement + " years)"


document.getElementById("growthSummaryText").innerText =
"To retire at age " + retirementAge +
" with an income equal to approximately " + growthFormatRand(futureMonthlyIncome) +
" per month, increasing by " + (incomeIncreaseRetirement * 100).toFixed(1) +
"% per year until age " + planningAge +
", you may need approximately " + growthFormatRand(capitalRequired) +
" in retirement capital. Based on your current retirement capital of " +
growthFormatRand(currentCapital) +
" and current monthly savings of " + growthFormatRand(monthlySavings) +
", increasing by " + (savingsIncrease * 100).toFixed(1) +
"% per year, you may need an estimated investment growth rate of " +
requiredGrowthRate.toFixed(2) +
"% per year before retirement."


resultsBox.style.display = "block"
}
</script>

function taxFormatRand(value) {
return "R " + Math.round(value).toLocaleString("en-ZA");
}


function taxGetNumber(id) {
return parseFloat(document.getElementById(id).value);
}


function calculateAnnualTaxBeforeRebate(taxableIncome) {
if (taxableIncome <= 245100) {
return taxableIncome * 0.18;
} else if (taxableIncome <= 383100) {
return 44118 + (taxableIncome - 245100) * 0.26;
} else if (taxableIncome <= 530200) {
return 79998 + (taxableIncome - 383100) * 0.31;
} else if (taxableIncome <= 695800) {
return 125599 + (taxableIncome - 530200) * 0.36;
} else if (taxableIncome <= 887000) {
return 185215 + (taxableIncome - 695800) * 0.39;
} else if (taxableIncome <= 1878600) {
return 259783 + (taxableIncome - 887000) * 0.41;
} else {
return 666339 + (taxableIncome - 1878600) * 0.45;
}
}


function calculateRebate(age) {
const primaryRebate = 17820;
const secondaryRebate = 9765;
const tertiaryRebate = 3249;


let rebate = primaryRebate;


if (age >= 65) {
rebate += secondaryRebate;
}


if (age >= 75) {
rebate += tertiaryRebate;
}


return rebate;
}


function calculateIncomeTax() {
const monthlyIncome = taxGetNumber("taxMonthlyIncome");
const age = taxGetNumber("taxAge");


const errorBox = document.getElementById("taxError");
const resultsBox = document.getElementById("taxResults");


errorBox.style.display = "none"
resultsBox.style.display = "none"


if (isNaN(monthlyIncome) || isNaN(age)) {
errorBox.innerText = "Please complete all fields."
errorBox.style.display = "block"
return;
}


if (monthlyIncome < 0) {
errorBox.innerText = "Monthly income cannot be negative."
errorBox.style.display = "block"
return;
}


if (age <= 0) {
errorBox.innerText = "Age must be greater than zero."
errorBox.style.display = "block"
return;
}


const annualIncome = monthlyIncome * 12;
const taxBeforeRebate = calculateAnnualTaxBeforeRebate(annualIncome);
const rebateApplied = calculateRebate(age);
const annualTaxPayable = Math.max(0, taxBeforeRebate - rebateApplied);
const monthlyPAYE = annualTaxPayable / 12;
const netMonthlyPay = monthlyIncome - monthlyPAYE;
const effectiveTaxRate = annualIncome > 0 ? (annualTaxPayable / annualIncome) * 100 : 0;


document.getElementById("taxAnnualPayable").innerText = taxFormatRand(annualTaxPayable);
document.getElementById("taxMonthlyPAYE").innerText = taxFormatRand(monthlyPAYE);
document.getElementById("taxNetMonthlyPay").innerText = taxFormatRand(netMonthlyPay);
document.getElementById("taxEffectiveRate").innerText = effectiveTaxRate.toFixed(1) + "%"
document.getElementById("taxAnnualIncome").innerText = taxFormatRand(annualIncome);
document.getElementById("taxRebateApplied").innerText = taxFormatRand(rebateApplied);


document.getElementById("taxSummaryText").innerText =
"Based on a gross monthly income of " + taxFormatRand(monthlyIncome) +
", your annual gross income is approximately " + taxFormatRand(annualIncome) +
". After applying the 2026/27 individual income tax tables and an age-based rebate of " +
taxFormatRand(rebateApplied) +
", your estimated annual tax payable is " + taxFormatRand(annualTaxPayable) +
". This equals estimated monthly PAYE of " + taxFormatRand(monthlyPAYE) +
", leaving an estimated net monthly pay of " + taxFormatRand(netMonthlyPay) +
". Your estimated effective tax rate is " + effectiveTaxRate.toFixed(1) + "%."


resultsBox.style.display = "block"
}
</script>
